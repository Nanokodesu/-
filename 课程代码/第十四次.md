## 2.2 代码实现
本质是在同一个界面加载两个 Cesium
两个 div 叠加
```    
<div id="cesiumContainer"></div>
    <div id="eagleEye"></div>```
两个视图的样式设计

```        #cesiumContainer {
            width: 100%;
            height: 100%;
            margin: 0;
            padding: 0;
            overflow: hidden;
        ``
 
        #eagleEye {
            position: absolute;
            width: 25%;
            height: 30%;
            bottom: 2%;
            right: 2%;
            z-index: 999;
        }}`
两个视图的参数设置

```        // 主视图
        const viewer = new Cesium.Viewer("cesiumContainer", {
            geocoder: true,
            homeButton: true,
            sceneModePicker: true,
            baseLayerPicker: false,
            navigationHelpButton: false,
            animation: false,
            timeline: false,
            fullscreenButton: false,
            infoBox: false,
        ``)
 
        // 开启帧率
        viewer.scene.debugShowFramesPerSecond = true;
 
        // 移除默认图层
        const imageryLayers = viewer.imageryLayers;
        imageryLayers.remove(imageryLayers.get(0));
 
        // 高德卫星图
        const AmapBiasProvider = new Cesium.AMapImageryProvider({
            style: 'img', // style: img、elec、cva
            crs: 'WGS84', // 使用84坐标系，默认为：GCJ02
            maximumLevel: 18,
            minimumLevel: 1,
        })
        viewer.imageryLayers.addImageryProvider(AmapBiasProvider);
 
        // 鹰眼图
        const eagleEye = new Cesium.Viewer("eagleEye", {
            geocoder: false,
            homeButton: false,
            sceneModePicker: false,
            baseLayerPicker: false,
            navigationHelpButton: false,
            animation: false,
            timeline: false,
            fullscreenButton: false,
            infoBox: false,
        })
 
        // 隐藏cesium ion
        eagleEye._cesiumWidget._creditContainer.style.display = "none";
 
        eagleEye.imageryLayers.remove(eagleEye.imageryLayers.get(0));;
        eagleEye.imageryLayers.addImageryProvider(
            new Cesium.AMapImageryProvider({
                style: 'elec', // style: img、elec、cva
                crs: 'WGS84', // 使用84坐标系，默认为：GCJ02
                maximumLevel: 18,
                minimumLevel: 1,
            })
        );
 
        // 禁用鹰眼图的相机操作，保证鹰眼图只受到主视图的控制和影响
        // 旋转
        eagleEye.scene.screenSpaceCameraController.enableRotate = false;
        // 缩放
        eagleEye.scene.screenSpaceCameraController.enableZoom = false;
        // 平移
        eagleEye.scene.screenSpaceCameraController.enableTranslate = false;
        // 倾斜
        eagleEye.scene.screenSpaceCameraController.enableTilt = false;
        // 相机观看方向
        eagleEye.scene.screenSpaceCameraController.enableLook = false;
 
        // 监听主视图的相机变化，同步鹰眼图的相机
        function reViewer() {
            eagleEye.camera.flyTo({
                destination: viewer.camera.positionWC,
                orientation: {
                    heading: viewer.camera.heading,
                    pitch: viewer.camera.pitch,
                    roll: viewer.camera.roll,
                },
                duration: 0.0,
            });
        }
 
        // 当主视图相机变化时，调用函数更新鹰眼图的相机
        viewer.camera.percentageChanged = 0.01;                // 相机变化的灵敏程度，越小越灵敏
        viewer.camera.changed.addEventListener(reViewer);     // 监听相机变化
}`
```

        const videoElement = document.getElementById('myVideo');
        // 视频元素与模拟时钟同步，保持自动播放

        // 创建视频墙
        const videoWall = viewer.entities.add({
            name: 'videoWall',
            wall: {
                positions: Cesium.Cartesian3.fromDegreesArrayHeights([
                    116.33898343805134, 39.99990467160987, 30,
                    116.33953471145098, 39.99990518926386, 30,
                ]),
                minimumHeights: [0, 0],                     //最低高度为0m起
                material: videoElement,
                outline: true,
                shadows: Cesium.ShadowMode.ENABLED,
            },
        });
 
        // 创建贴地视频
        const videoGround = viewer.entities.add({
            name: 'videoGround',
            rectangle: {
                coordinates: Cesium.Rectangle.fromDegrees(
                    116.33714109411144, 40.0010804071651,
                    116.3381148055563, 40.001412600007086,
                ),
                material: videoElement,
                classificationType: Cesium.ClassificationType.BOTH,
            },
        });
 
        // 默认状态
        videoWall.show = false;
        videoGround.show = false;  
 
        // 切换视频
        const dropdown = document.getElementById('dropdown');
        function change() {
            switch (dropdown.value) {
                case 'option1':
                    videoWall.show = true;
                    videoGround.show = false;
                    viewer.zoomTo(videoWall);
                    break;
                case 'option2':
                    videoGround.show = true;
                    videoWall.show = false;
                    viewer.zoomTo(videoGround);
                    break;
            }
        }
        }

        // 滑动条调整函数
        function change() {
            // 获取等高距滑动条值
            const HS = document.getElementById("HeightSpa").value;
            // 等高距文本显示当前值
            document.getElementById("HeightSpaValue").value = HS;
            // 获取线宽滑动条值
            const LW = document.getElementById("LineWid").value;
            // 线宽文本显示当前值
            document.getElementById("LineWidValue").value = LW;
 
            // 创建等高线材质
            const material = Cesium.Material.fromType("ElevationContour");
            // 设置等高线材质属性
            const contourUniforms = {``;
            contourUniforms.width = LW;
            contourUniforms.spacing = HS;
            contourUniforms.color = Cesium.Color.RED;
            material.uniforms = contourUniforms;
            viewer.scene.globe.material = material;
        }
 
        // 文本框调整函数
        function HeightSpaValue() {
            const HS = document.getElementById("HeightSpaValue").value;
            document.getElementById("HeightSpa").value = HS;
            change();
        }
 
        function LineWidValue() {
            const LW = document.getElementById("LineWidValue").value;
            document.getElementById("LineWid").value = LW;
            change();
        }

            }
