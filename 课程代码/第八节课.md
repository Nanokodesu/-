<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="UTF-8">
    <meta http-equiv="X-UA-Compatible" content="IE=edge">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>交互绘制</title>
    <script type="text/javascript" src="./Build/Cesium/Cesium.js"></script>
    <link rel="stylesheet" href="./Build/Cesium/Widgets/widgets.css">
    <style>
        html,
        body,
        #cesiumContainer {
            width: 100%;
            height: 100%;
            margin: 0;
            padding: 0;
            overflow: hidden;
        }

        .toolbar {
            position: absolute;
            top: 10px;
            left: 10px;
            background: rgba(0, 0, 0, 0.5);
            padding: 10px;
            border-radius: 5px;
            color: white;
        }
    </style>
</head>

<body>
    <div id="cesiumContainer"></div>
    <div id="menu" class="toolbar">
        <select id="dropdown" onchange="edit()">
            <option value="null">默认状态</option>
            <option value="drawPoint">绘制点</option>
            <option value="drawLine">绘制线</option>
            <option value="drawPolygon">绘制面</option>
            <option value="drawCircle">绘制圆</option>
            <option value="drawRectangle">绘制矩形</option>
            <option value="drawModel">绘制模型</option>
            <option value="clear">清除</option>
        </select>
    </div>
    <script type="text/javascript">
        Cesium.Ion.defaultAccessToken = '自己的Token';

        const viewer = new Cesium.Viewer("cesiumContainer", {
            geocoder: true,
            homeButton: true,
            sceneModePicker: true,
            baseLayerPicker: false,
            navigationHelpButton: false,
            animation: true,
            timeline: true,
            fullscreenButton: false,
            infoBox: false,
        })

        // 隐藏cesium ion
        viewer._cesiumWidget._creditContainer.style.display = "none";

        // 开启帧率
        viewer.scene.debugShowFramesPerSecond = true;

        // SHP转3DTiles
        (async function () {
            var tileset = await Cesium.Cesium3DTileset.fromUrl(
                "./Data/SHP/Slice_shp/tileset.json",
            );
            viewer.scene.primitives.add(tileset);
            viewer.flyTo(tileset);
            return tileset;
        })();

        var drawingMode;                  // 绘制模式
        var floatingPoint;                // 存储第一个点并判断是否开始获取鼠标移动结束位置
        var activeShapePoints = [];       // 存储动态点数组
        var activeShape;                  // 存储动态图形
        
        // 绘制点
        function drawPoint(position) {
            const addPoint = {
                name: '点',
                position: position,
                point: {
                    color: Cesium.Color.BLUE,
                    pixelSize: 5,
                    outlineWidth: 2,
                },
            };
            viewer.entities.add(addPoint);
            return addPoint;
        };

        // 添加模型
        function drawModel(position) {
            const addModel = {
                name: '模型',
                position: position,
                model: {
                    uri: "./Data/CZML/GroundVehicle.glb",
                    scale: 2.0,               
                    minimumPixelSize: 128,    
                    maximumScale: 10,         
                },
            };
            viewer.entities.add(addModel);
            return addModel;
        };

        // 绘制图形
        function drawShape(positionData) {
            var shape;
            if (drawingMode === 'line') {
                shape = viewer.entities.add({
                    name: '线',
                    polyline: {
                        positions: positionData,
                        width: 5,
                        material: Cesium.Color.RED,
                        clampToGround: true,
                    },
                });
            } else if (drawingMode === 'polygon') {
                shape = viewer.entities.add({
                    name: '面',
                    polygon: {
                        hierarchy: positionData,
                        material: Cesium.Color.AZURE.withAlpha(0.8),
                    },
                });
            }
            else if (drawingMode === 'circle') {
                // 判断positionData是否为回调函数，如果是则调用getValue(0)，用于动态绘制图形
                var value = typeof positionData.getValue === 'function' ? positionData.getValue(0) : positionData;

                shape = viewer.entities.add({
                    position: activeShapePoints[0],
                    ellipse: {
                        semiMinorAxis: new Cesium.CallbackProperty(
                            function () {
                                // 计算两点之间的距离：半径
                                var r = Math.sqrt(
                                    Math.pow(value[0].x - value[value.length - 1].x, 2) +
                                    Math.pow(value[0].y - value[value.length - 1].y, 2)
                                );
                                // 确保椭圆的短半轴有一个有效的、非负的值
                                return r;
                            }, false),
                        semiMajorAxis: new Cesium.CallbackProperty(
                            function () {
                                // 计算两点之间的距离：半径
                                var r = Math.sqrt(Math.pow(value[0].x - value[value.length - 1].x, 2) +
                                    Math.pow(value[0].y - value[value.length - 1].y, 2));
                                return r;
                            },
                            false),
                        material: Cesium.Color.CHOCOLATE.withAlpha(0.8),
                    },
                });
            }
            else if (drawingMode === 'rectangle') {
                var arrange = typeof positionData.getValue === 'function' ? positionData.getValue(0) : positionData;
                shape = viewer.entities.add({
                    name: '矩形',
                    rectangle: {
                        coordinates: new Cesium.CallbackProperty(function () {
                            var object = Cesium.Rectangle.fromCartesianArray(arrange);
                            return object;
                        }, false),
                        material: Cesium.Color.CORNFLOWERBLUE.withAlpha(0.8),
                    },
                });
            }
            return shape;
        }

        // 左键按下事件用于选择点位
        const handler = new Cesium.ScreenSpaceEventHandler(viewer.scene.canvas);
        // 注册鼠标左键事件
        handler.setInputAction(function (event) {
            const earthPosition = viewer.scene.pickPosition(event.position);
            if (drawingMode == "point") {
                drawPoint(earthPosition);
            }
            else if (drawingMode == "model") {
                drawModel(earthPosition);
            }
            else if (drawingMode == "line" || drawingMode == "polygon" || drawingMode == "circle" || drawingMode == "rectangle") {
                if (Cesium.defined(earthPosition)) {
                    // 第一次单击时，创建一个点
                    if (activeShapePoints.length === 0) {
                        activeShapePoints.push(earthPosition);
                        floatingPoint = drawPoint(earthPosition);

                        var dynamicPositions = new Cesium.CallbackProperty(function () {
                            if (drawingMode === 'polygon') {
                                return new Cesium.PolygonHierarchy(activeShapePoints);
                            } else {
                                return activeShapePoints;
                            }
                        }, false);

                        activeShape = drawShape(dynamicPositions);   // 绘制动态模型
                    };

                    // 第二次及后续点击时，添加点
                    // 由于 dynamicPositions 是一个 Cesium.CallbackProperty，
                    // 它会根据 activeShapePoints 数组的变化实时更新 activeShape 的形状。
                    // 因此，每次点击时，activeShape 都会根据 activeShapePoints 数组中的点重新渲染。
                    
                    activeShapePoints.push(earthPosition);
                    drawPoint(earthPosition);
                }
            }
        }, Cesium.ScreenSpaceEventType.LEFT_CLICK);

        // 注册鼠标移动事件
        handler.setInputAction(function (event) {
            if (Cesium.defined(floatingPoint)) {
                // 获取鼠标移动的最后位置
                const newPosition = viewer.scene.pickPosition(event.endPosition);
                if (Cesium.defined(newPosition)) {
                    // 动态去除数组的最后一个点，否则线无法画直
                    activeShapePoints.pop();
                    // 只添加末尾点，保证只保留鼠标的位置
                    activeShapePoints.push(newPosition);
                }
            }
        }, Cesium.ScreenSpaceEventType.MOUSE_MOVE);

        // 注册鼠标右键事件
        handler.setInputAction(function (event) {
            activeShapePoints.pop();                       // 移除最后一个点
            if (activeShapePoints.length > 1) {
                drawShape(activeShapePoints);              // 绘制图形（注意与动态图形的差异）
            }
            viewer.entities.remove(floatingPoint);         // 移除第一个点
            viewer.entities.remove(activeShape);           // 移除动态图形
            floatingPoint = undefined;
            activeShape = undefined;
            activeShapePoints = [];                        // 清空动态点数组
        }, Cesium.ScreenSpaceEventType.RIGHT_CLICK);

        // 指定绘制模式
        const dropdown = document.getElementById('dropdown');
        function edit() {
            switch (dropdown.value) {
                case 'null':
                    drawingMode = 'null';
                    break;
                case 'drawPoint':
                    drawingMode = 'point';
                    break;
                case 'drawLine':
                    drawingMode = 'line';
                    break;
                case 'drawPolygon':
                    drawingMode = 'polygon';
                    break;
                case 'drawCircle':
                    drawingMode = 'circle';
                    break;
                case 'drawRectangle':
                    drawingMode = 'rectangle';
                    break;
                case 'drawModel':
                    drawingMode = 'model';
                    break;
                case 'clear':
                    viewer.entities.removeAll();
                    break;
                default:
                    break;
            }
        }

    </script>
</body>

</html>

        const viewer = new Cesium.Viewer("cesiumContainer", {
            geocoder: true,
            homeButton: true,
            sceneModePicker: true,
            baseLayerPicker: false,
            navigationHelpButton: false,
            animation: true,

            fullscreenButton: false,
            infoBox: true,          // 需要开启信息框(潜在sandbox限制问题，但不影响前端显示)

            );
            viewer.scene.primitives.add(tileset);
            viewer.flyTo(tileset);
        })();

        // 被选择建筑的信息
        const selected = {
            feature: undefined,
            originalColor: new Cesium.Color(),
        }

        // 即时选择的轮廓颜色
        const silhouetteYellow = Cesium.PostProcessStageLibrary.createEdgeDetectionStage();
        silhouetteYellow.uniforms.color = Cesium.Color.YELLOW;
        silhouetteYellow.uniforms.length = 0.01;
        silhouetteYellow.selected = [];

        //点击选择的轮廓颜色
        const silhouetteBlue = Cesium.PostProcessStageLibrary.createEdgeDetectionStage();
        silhouetteBlue.uniforms.color = Cesium.Color.BLUE;
        silhouetteBlue.uniforms.length = 0.01;
        silhouetteBlue.selected = [];

        // 添加为渲染后处理阶段
        viewer.scene.postProcessStages.add(
            Cesium.PostProcessStageLibrary.createSilhouetteStage([
                silhouetteYellow,
                silhouetteBlue,

        );

        // 创建Div，用于设置建筑名称显示的样式（Mouse_move）
        const div = document.createElement("div");
        viewer.container.appendChild(div);

        div.style.display = "none";

                return
            } else {

                div.style.top = `${position.y}px`;                      
                div.style.left = `${position.x + 15}px`;                        // 加15是为了避免遮挡
                div.textContent = pickedFeature.getProperty("RefName");
            }

        const selectedEntity = new Cesium.Entity();

        // 定义点击储存建筑所有信息的函数

        function MouseclickInfo(pickedFeature) {
            const description =
                `<table class="cesium-infoBox-defaultTable"><tbody>
                <tr><th>建筑名称</th><td>` + 
                `${pickedFeature.getProperty("RefName")}</td></tr>` +
                `<tr><th>建筑类型</th><td>${pickedFeature.getProperty("Type")}</td></tr>` +
                `<tr><th>建筑高度/m</th><td>${pickedFeature.getProperty("Elevation")}</td></tr>` +
                `<tr><th>数据更新时间</th><td>${pickedFeature.getProperty("DocUpdate")}</td></tr>` +
                `</tbody></table>`;
            return description;
        }

        // 即时显示操作
        const handler = new Cesium.ScreenSpaceEventHandler(viewer.scene.canvas);
        handler.setInputAction(function (event) {

            // 如果前一个模型高亮了，则取消高亮
            silhouetteYellow.selected = [];

            // 选择新对象
            const pickedObject = viewer.scene.pick(event.endPosition);

            // 对新对象黄色高亮显示，并且显示名称
            if (!Cesium.defined(pickedObject)) {
                // div.style.display = "none";
                return;
            }
            else {
                MousemoveName(pickedObject, event.endPosition);
                silhouetteYellow.selected = [pickedObject];
            }
        }, Cesium.ScreenSpaceEventType.MOUSE_MOVE);

        // 点击显示操作
        handler.setInputAction(function (event) {

            // 如果前一个模型高亮了，则取消高亮
            silhouetteBlue.selected = [];

            // 选择新对象
            const pickedObject = viewer.scene.pick(event.position);
            if (!Cesium.defined(pickedObject)) {
                return;
            } else {
                // 避免与黄色高亮冲突
                if (silhouetteYellow.selected.length > 0) {
                    const highlightedFeature = silhouetteYellow.selected[0];
                    if (pickedObject === highlightedFeature) {
                        silhouetteYellow.selected = [];
                    }
                }
                // 对新对象蓝色高亮显示，并且显示信息
                silhouetteBlue.selected = [pickedObject];
                selectedEntity.name = "北京林业大学及周边建筑信息";
                selectedEntity.description = MouseclickInfo(pickedObject);
                viewer.selectedEntity = selectedEntity;
            }
        }, Cesium.ScreenSpaceEventType.LEFT_CLICK);

2.1 关键 API

    Cesium.PostProcessStageLibrary.createEdgeDetectionStage()


反引号
- 默认换行（多行字符串的显示）
- 可以更好进行引用（模板字符串）
- 即是⬇️⬇️⬇️
