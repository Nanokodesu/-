~~~高德矢量地图
<body>
  <div id="cesiumContainer"></div>
  <script>
    Cesium.Ion.defaultAccessToken = '自己的Token';
    const viewer = new Cesium.Viewer(
        "cesiumContainer", {
        geocoder: true,
        homeButton: true,
        sceneModePicker: true,
        baseLayerPicker: false,
        navigationHelpButton: false,
        animation: false,
        timeline: false,
        fullscreenButton: false,
    }
    )
 
    // 移除默认图层
    const imageryLayers = viewer.imageryLayers;
    imageryLayers.remove(imageryLayers.get(0));
 
    // 采用高德矢量图层
    const AmapProvider = new Cesium.UrlTemplateImageryProvider({
        url: 'https://webrd04.is.autonavi.com/appmaptile?lang=zh_cn&size=1&scale=1&style=7&x={x}&y={y}&z={z}',
        maximumLevel: 18,
        minimumLevel: 1,
        credit: 'Amap',
    })
    viewer.imageryLayers.addImageryProvider(AmapProvider);
  </script>
</body>

~~~

天地图
- 申请天地图官网免费Key：[https://www.tianditu.gov.cn/](https://www.tianditu.gov.cn/ "https://www.tianditu.gov.cn/")
- 【开发资源】-【地图API】
![](https://p.ananas.chaoxing.com/star3/origin/b8d13535692e5bbc1b3ef106bb309e01.png "image21.png")

```JavaScript
// 采用天地图
const TianDiKey = "a4e9389a79a5674b8676eeaff6cb6b01";
const TianDiProvider = new Cesium.WebMapTileServiceImageryProvider({
  url:
    "http://t{s}.tianditu.com/img_w/wmts?service=wmts&request=GetTile&version=1.0.0&LAYER=img&tileMatrixSet=w&TileMatrix={TileMatrix}&TileRow={TileRow}&TileCol={TileCol}&style=default&format=tiles&tk=" +
    TianDiKey,
  subdomains: ["0", "1", "2", "3", "4", "5", "6", "7"],
  layer: "tdtImgLayer",
  style: "default",
  format: "image/jpeg",
  tileMatrixSetID: "GoogleMapsCompatible",
});
viewer.imageryLayers.addImageryProvider(TianDiProvider);
```


~~~叠加图层
// 移除默认图层
        const imageryLayers = viewer.imageryLayers;
        imageryLayers.remove(imageryLayers.get(0));
// 采用高德+OSM图源
        const AmapProvider = new Cesium.UrlTemplateImageryProvider({
            url: 'https://webst01.is.autonavi.com/appmaptile?style=6&x={x}&y={y}&z={z}',
        })
        const OSMProvider = new Cesium.OpenStreetMapImageryProvider({
            url: 'https://a.tile.openstreetmap.org/',
        })
 
        viewer.imageryLayers.addImageryProvider(AmapProvider);
        const addMap = viewer.imageryLayers.addImageryProvider(OSMProvider);
        addMap.alpha = 0.5;
~~~
