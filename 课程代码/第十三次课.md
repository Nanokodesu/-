- 粒子特效设置
    

```JavaScript
 // 粒子特效位置
const position = Cesium.Cartesian3.fromDegrees(116.34083842367593, 40.00234978089876, 1000.0);
const snowModelMatrix = new Cesium.Transforms.eastNorthUpToFixedFrame(position);        // 盒装发射器需要局地坐标系

// 下雪天，粒子特效相关函数定义
const snowGravityScratch = new Cesium.Cartesian3();                             // 实时更新粒子位置
function snowUpdate(particle) {
    Cesium.Cartesian3.normalize(particle.position, snowGravityScratch);         // 标准化粒子位置

    Cesium.Cartesian3.multiplyByScalar(
        snowGravityScratch,
        Cesium.Math.randomBetween(-100.0, -30.0),                              // 随机速度, 负值表示位置下降
        snowGravityScratch
    );

    // 粒子位置根据snowGravityScratch速度更新
    Cesium.Cartesian3.add(
        particle.position,
        snowGravityScratch,
        particle.position
    );
}

// 下雪天，粒子特效相关常量定义
const snowBoxEmitter = new Cesium.BoxEmitter(new Cesium.Cartesian3(10000.0, 10000.0, 10.0));
const minimumSnowImageSize = new Cesium.Cartesian2(10, 10);                      // 最小雪花尺寸
const maximumSnowImageSize = new Cesium.Cartesian2(20, 20);                      // 最大雪花尺寸

// 下雪天配置项
const snowOption = {
    modelMatrix: snowModelMatrix,
    lifeTime: 15.0,                                                // 粒子生命周期（秒）
    emitter: snowBoxEmitter,                                       // 粒子发射类型，盒装发射器
    startScale: 0.5,                                               // 粒子生命周期开始时的比例
    endScale: 1.0,                                                 // 粒子生命周期结束时的比例
    image: './Texture/Snow.png',                                   // 粒子图片
    emissionRate: 5000.0,                                          // 每秒发射的粒子数
    startColor: Cesium.Color.WHITE.withAlpha(0.0),                 // 粒子生命周期开始时的颜色
    endColor: Cesium.Color.WHITE.withAlpha(1.0),                   // 粒子生命周期结束时的颜色
    minimumImageSize: minimumSnowImageSize,                        // 最小雪花尺寸
    maximumImageSize: maximumSnowImageSize,                        // 最大雪花尺寸
    updateCallback: snowUpdate,                                    // 每一帧都要调用一次回调函数以更新粒子
}

// 下雨天，粒子特效相关函数定义
const rainModelMatrix = new Cesium.Matrix4.fromTranslation(position);           // 球形发射器不需要局地坐标系

const rainGravityScratch = new Cesium.Cartesian3();                             // 实时更新粒子位置
function rainUpdate(particle) {
    Cesium.Cartesian3.normalize(particle.position, rainGravityScratch);         // 标准化粒子位置

    Cesium.Cartesian3.multiplyByScalar(
        rainGravityScratch,
        -1000,                                   // 雨水比雪花下降速快，所以负值更小一些
        rainGravityScratch
    );
    // 粒子位置根据rainGravityScratch速度更新
    Cesium.Cartesian3.add(
        particle.position,
        rainGravityScratch,
        particle.position
    );
}

// 下雨天配置项
const rainOption = {
    modelMatrix: rainModelMatrix,
    lifeTime: 15.0,                                                // 粒子生命周期（秒）
    emitter: new Cesium.SphereEmitter(10000.0),                    // 粒子发射类型，下雨范围
    startScale: 1.0,                                               // 粒子生命周期开始时的比例
    endScale: 0.0,                                                 // 粒子生命周期结束时的比例
    image: './Texture/Rain.png',                                   // 粒子图片
    emissionRate: 15000.0,                                          // 每秒发射的粒子数
    startColor: Cesium.Color.WHITE.withAlpha(0.5),                 // 粒子生命周期开始时的颜色
    endColor: Cesium.Color.WHITE.withAlpha(0.98),                   // 粒子生命周期结束时的颜色
    imageSize: new Cesium.Cartesian2(10, 10),                        // 每个雨滴的尺寸
    updateCallback: rainUpdate,                                     // 每一帧都要调用一次回调函数以更新粒子
}

let snowParticleSystem = null;
let rainParticleSystem = null;

const dropdown = document.getElementById('dropdown');
function edit() {
    if (dropdown.value === "null") {
        if (snowParticleSystem) {
            viewer.scene.primitives.remove(snowParticleSystem);
            snowParticleSystem = null;
        }
        if (rainParticleSystem) {
            viewer.scene.primitives.remove(rainParticleSystem);
            rainParticleSystem = null;
        }
    } else if (dropdown.value === "snow") {
        if (rainParticleSystem) {
            viewer.scene.primitives.remove(rainParticleSystem);
            rainParticleSystem = null;
        }
        snowParticleSystem = new Cesium.ParticleSystem(snowOption);
        viewer.scene.primitives.add(snowParticleSystem);
    } else if (dropdown.value === "rain") {
        if (snowParticleSystem) {
            viewer.scene.primitives.remove(snowParticleSystem);
            snowParticleSystem = null;
        }
        rainParticleSystem = new Cesium.ParticleSystem(rainOption);
        viewer.scene.primitives.add(rainParticleSystem);
    }
}
```

        // 粒子起点，发射源
        const staticPosition = Cesium.Cartesian3.fromDegrees(116.33915181764083, 40.0059357901581, 0);
 
        // 将粒子系统的模型坐标系转变为世界坐标系的矩阵
        const modelMatrix = Cesium.Transforms.eastNorthUpToFixedFrame(staticPosition);
 
        // 粒子发射器的模型矩阵
        function computeEmitterModelMatrix() {
            const hpr = new Cesium.HeadingPitchRoll.fromDegrees(0.0, 0.0, 0.0);              // 粒子发射器的方向、俯仰角、滚动角
            const trs = new Cesium.TranslationRotationScale();                               // 定义一个位移、旋转、缩放
            trs.translation = Cesium.Cartesian3.fromElements(0.0, 0.0, 0.0);                 // 发射器在x y z轴的平移位置
            trs.rotation = Cesium.Quaternion.fromHeadingPitchRoll(hpr);                      // 不旋转
            return Cesium.Matrix4.fromTranslationRotationScale(trs, new Cesium.Matrix4());         // 返回一个模型矩阵;
        }
        
        viewer.scene.primitives.add(
            new Cesium.ParticleSystem({
                image: "./Texture/Fire.png",
                startColor: Cesium.Color.RED.withAlpha(0.8),
                endColor: Cesium.Color.YELLOW.withAlpha(0.3),
                startScale: 2.0,
                endScale: 6.0,
                minimumParticleLife: 1.0,
                maximumParticleLife: 4.0,
                lifetime: 180.0,                                              // 整个粒子系统的生命周期（秒），配合loop参数使用，表示粒子系统是否循环播放
                loop: false,
                speed: 5.0,
                imageSize: new Cesium.Cartesian2(10, 10),
                emitter: new Cesium.CircleEmitter(10.0),
                emissionRate: 20.0,
                modelMatrix: modelMatrix, 
                emitterModelMatrix: computeEmitterModelMatrix(),
            })
        );

```JavaScript
// 打印图片
function savePng (data, pngName){
    const savelink = document.createElement('a');                 // 创建下载链接标签<a>
    savelink.href = data;                                         // 设置下载链接
    savelink.download = pngName;                                  // 设置下载文件名
    document.body.appendChild(savelink);                          // 将下载链接标签<a>添加到body中
    savelink.click();                                             // 模拟点击下载链接标签<a>
}

// 打印当前场景
function printScreenScene() {
    viewer.render();                                                    // 渲染当前场景
    const image = viewer.scene.canvas.toDataURL('image/png');           // 获取下载链接
    savePng(image, '当前场景');
}   

```
