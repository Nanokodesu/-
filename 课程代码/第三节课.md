**CZML**
CZML是一种用来描述动态场景的JSON架构的语言，描述了场景中的单个对象（如单个飞机、汽车）的图形属性
官方提供了多种小模型供使用：cesium/Apps/SampleData/models at main
CZML的关键属性：Packet · AnalyticalGraphicsInc/czml

```
	// CZML静态数据
const czml = [    //有这么一个点
  {
    id: "document",
    name: "CZML Model",
    version: "1.0",
  },
  {
    id: "aircraft model",
    name: "Cesium Air",
    position: {
      cartographicDegrees: [116.342261, 40.001303, 150],  //具体位置
    },
    model: {
      gltf: "./Data/CZML/Cesium_Air.glb",
      scale: 2.0,
      minimumPixelSize: 128,  //大小设置
    },
  },
];
const czmlData = Cesium.CzmlDataSource.load(czml);
viewer.dataSources.add(czmlData);
viewer.trackedEntity = czmlData.entities.getById("aircraft model");

// CZML动态轨迹
const czml = [
  {
    // Packet 1//地球运行的时间
    id: "document",
    name: "CZML Air - Time Dynamic",
    version: "1.0",
    clock: {
      interval: "2023-12-12T07:00:00Z/2023-12-12T07:10:00Z",
      currentTime: "2023-12-12T07:00:00Z",//不是飞机的时间
      multiplier: 10,//倍速
    },
  },
  {
    // Packet 2
    id: "Aircraft",
    availability: "2023-12-12T07:00:00Z/2023-12-12T07:10:00Z", //T表示分隔符，Z表示的是UTC
    model: {
      gltf: "./Data/CZML/Cesium_Air.glb",
      scale: 2.0,
      minimumPixelSize: 128,
    },
    path: {
      material: {
        polylineOutline: {
          color: { rgba: [255, 0, 255, 127] },//=0-255
          outlineColor: { rgba: [255, 0, 255, 127] },
          outlineWidth: 3,
        },
      },
      width: 5,
      leadTime: 5,
      trailTime: 1000,//保存的时间
      resolution: 5,
    },
    position: {//位置信息
      epoch: "2023-12-12T07:00:00Z",
      cartographicDegrees: [
        0, 116.342261, 40.001303, 150, 200, 116.323869, 40.001303, 150, 400, 116.303469, 39.9933,//位置设定
        150, 600, 116.308833, 39.970639, 150,
      ],
    },
  },
];
Cesium.CzmlDataSource.load(czml)
  .then((dataSource) => {
    viewer.dataSources.add(dataSource);
    const aircraftEntity = dataSource.entities.getById("Aircraft");
    if (aircraftEntity) {
      viewer.trackedEntity = aircraftEntity;
      // 设置模型可以随路径方向转向
      aircraftEntity.orientation = new Cesium.VelocityOrientationProperty(aircraftEntity.position);
      aircraftEntity.model.alignedAxis = new Cesium.VelocityVectorProperty(
        aircraftEntity.position,
        true,
      );
    } else {
      console.warn("Aircraft entity not found.");
    }
  })
  .catch((error) => {
    console.error("Failed to load CZML data: ", error);
  });
  
```
