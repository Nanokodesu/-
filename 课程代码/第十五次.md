 let tileset;
(async function () {
    tileset = await Cesium.Cesium3DTileset.fromUrl(
        "./Data/SHP/Slice_shp/tileset.json"
    );
    viewer.scene.primitives.add(tileset);

// 初始化可视域参数 
let frustrumLabel = null;                 // 可视域分析中的鼠标提示
let viewPointFlag = false;                // 是否选择了视点
let pickPoints = [];                      //被选中的点
let pickPositions = [];                   // 被选中的点的坐标
let boardLines = [];                      //可视区域边界线
let activeLine;                           //视野方向线

let handler;                              // 定义全局变量 handler

// 添加鼠标左键处理事件
function setBuildFrustrumHandler(flag) {
    if (flag) {
        if (!handler) {
            handler = new Cesium.ScreenSpaceEventHandler(viewer.scene.canvas);
        }
        handler.setInputAction(function (event) {
            const earthPosition = viewer.scene.pickPosition(event.position);
            if (Cesium.defined(earthPosition)) {
                pickPoints.push(createPoint(earthPosition));
                if (pickPositions.length > 1) {
                    // 进行可视化分析
                    // 先清除边界线
                    for (let i = 0; i < boardLines.length; ++i) {
                        viewer.entities.remove(boardLines[i]);
                    }
                    frustrumLabel.label.text = "可视域分析中...";
                    // 分析完毕清除鼠标事件
                    setBuildFrustrumHandler(false);
                    // 进行可视域分析
                    viewAreaAnalysis(45, pickPositions[0], pickPositions[1]);
                } else {
                    pickPositions.push(earthPosition);
                    viewPointFlag = true;
                    const dynamicPositions = new Cesium.CallbackProperty(function () {
                        return pickPositions;
                    }, false);
                    pickPositions.push(earthPosition);                      // 初始化一条长度为零的线段（起点和终点相同）
                    activeLine = drawLine(dynamicPositions, Cesium.Color.WHITE, Cesium.Color.WHITE);
                }
            }
        }, Cesium.ScreenSpaceEventType.LEFT_CLICK);
        //添加鼠标移动处理事件
        handler.setInputAction(function (event) {
            const newPosition = viewer.scene.pickPosition(event.endPosition);
            if (Cesium.defined(newPosition)) {
                if (frustrumLabel == null) {
                    frustrumLabel = createLabel(newPosition, "点击选择视点");
                }
                else {
                    frustrumLabel.position = newPosition;
                    if (viewPointFlag == true) {
                        frustrumLabel.label.text = "点击视线方向";
                        pickPositions.pop();                           // 删除初始化的终点坐标
                        pickPositions.push(newPosition);
                        if (boardLines.length > 1) {
                            for (let i = 0; i < boardLines.length; ++i) {
                                viewer.entities.remove(boardLines[i]);
                            }
                        }
                        //构建可视区域
                        if (pickPositions.length > 1) {
                            boardLines = drawSector(pickPositions[0], pickPositions[1]);
                        }
                    }
                }
            }
        }, Cesium.ScreenSpaceEventType.MOUSE_MOVE);
    } else {
        handler.removeInputAction(Cesium.ScreenSpaceEventType.MOUSE_MOVE);
        handler.removeInputAction(Cesium.ScreenSpaceEventType.LEFT_CLICK);

// 扇形可视区域的两条边界线（45度到135度）
function drawSector(startPoint, endPoint) {
    const lines = [];
    const leftLine = rotateLine(Cesium.Math.toRadians(45), startPoint, endPoint);
    const rightLine = rotateLine(Cesium.Math.toRadians(-45), startPoint, endPoint);
    lines.push(leftLine);
    lines.push(rightLine);
    return lines;
}
function rotateLine(radian, startPoint, endPoint) {
    const position_Cartesian3 = rotatePoint(radian, startPoint, endPoint);
    const LinePoints = [];
    LinePoints.push(startPoint);
    LinePoints.push(position_Cartesian3);
    const line = drawLine(LinePoints, Cesium.Color.YELLOW, Cesium.Color.YELLOW);
    return line;
}

//计算某一点绕另一点旋转radian后的终点坐标
function rotatePoint(radian, startPoint, endPoint) {
    // 获取视线的方向向量 (终点 - 起点)
    const direction = Cesium.Cartesian3.subtract(endPoint, startPoint, new Cesium.Cartesian3());

    // 获取视点在地表的法向量 (向上向量)
    const up = viewer.scene.globe.ellipsoid.geodeticSurfaceNormal(startPoint);

    // 右手法则，计算垂直于视线的向右向量
    const right = new Cesium.Cartesian3();
    Cesium.Cartesian3.cross(direction, up, right);
    Cesium.Cartesian3.normalize(right, right);

    // 计算倾斜扇面的垂直轴，即旋转轴
    const viewUp = new Cesium.Cartesian3();
    Cesium.Cartesian3.cross(right, direction, viewUp);
    Cesium.Cartesian3.normalize(viewUp, viewUp); 

    // 基于旋转值和旋转轴构建四元数，并将四元数转换为旋转矩阵
    const quaternion = Cesium.Quaternion.fromAxisAngle(viewUp, radian);
    const rotationMatrix = Cesium.Matrix3.fromQuaternion(quaternion);

    // 将视线方向向量乘以旋转矩阵，得到旋转后的方向向量
    const rotatedDirection = new Cesium.Cartesian3();
    Cesium.Matrix3.multiplyByVector(rotationMatrix, direction, rotatedDirection);

    // 计算最终坐标 = 起点坐标 + 旋转后的方向向量
    const resultPoint = new Cesium.Cartesian3();
    Cesium.Cartesian3.add(startPoint, rotatedDirection, resultPoint);

    return resultPoint;
}

// 可视化分析
function viewAreaAnalysis(degree, startPoint, endPoint) {
    for (let i = -degree; i <= degree; ++i) {
        const radian = Cesium.Math.toRadians(i);
        const destPoint = rotatePoint(radian, startPoint, endPoint);
        getIntersectPoint(startPoint, destPoint);
    }
    viewer.entities.remove(frustrumLabel);
    viewer.entities.remove(activeLine);
    pickPositions = [];
    for (let i = 0; i < pickPoints.length; ++i) {
        viewer.entities.remove(pickPoints[i]);
    }
    pickPoints = [];
}

// 计算两点连成的直线段与地形或者建筑的交点，并绘制可视线
function getIntersectPoint(startPoint, endPoint) {
    // 用向量标准化方式明确两点连线的方向
    const direction = Cesium.Cartesian3.normalize(Cesium.Cartesian3.subtract(endPoint, startPoint, new Cesium.Cartesian3()), new Cesium.Cartesian3());
    // 建立射线
    const ray = new Cesium.Ray(startPoint, direction);
    // 计算相交点
    const result = viewer.scene.pickFromRay(ray);
    if (Cesium.defined(result)) {
        const intesectPosition = result.position;
        //判断相交点是否比终点更远
        if (distanceBetweenTwoPoints(startPoint, endPoint) > distanceBetweenTwoPoints(intesectPosition, startPoint)) {
            drawLine([startPoint, intesectPosition], Cesium.Color.GREEN, Cesium.Color.GREEN);
            drawLine([intesectPosition, endPoint], Cesium.Color.RED, Cesium.Color.RED);
        } else {
            drawLine([startPoint, endPoint], Cesium.Color.GREEN, Cesium.Color.GREEN);
        }
    } else {
        drawLine([startPoint, endPoint], Cesium.Color.GREEN, Cesium.Color.GREEN);
    }
}

// 绘制视点
function createPoint(position) {
    const addPoint = {
        name: '点',
        position: position,
        point: {
            color: Cesium.Color.WHITE,
            pixelSize: 3,
            outlineWidth: 1,
        },
    };
    viewer.entities.add(addPoint);
    return addPoint;
};

// 绘制视野连线
function drawLine(positionData, material, depthFailMaterial) {
    const shape = viewer.entities.add({
        name: '线',
        polyline: {
            positions: positionData,
            arcType: Cesium.ArcType.NONE,
            width: 5,
            material: material,
            depthFailMaterial: depthFailMaterial,                 //被地形和建筑遮挡到的部分
        },
    });
    return shape;
}

// 绘制标签
function createLabel(position, text) {
    return viewer.entities.add({
        name: '标签',
        position: Cesium.Cartesian3.fromDegrees(position.longitude, position.latitude, position.height),
        label: {
            text: text,
            font: '8pt monospace',
            style: Cesium.LabelStyle.FILL_AND_OUTLINE,
            outlineWidth: 2,
            verticalOrigin: Cesium.VerticalOrigin.BOTTOM,                   // 垂直方向以底部对齐
            pixelOffset: new Cesium.Cartesian2(0, -9)
        }
    });
}

// 计算两点之间的距离
function distanceBetweenTwoPoints(point1, point2) {
    // 计算两个点之间的差向量
    const diff = Cesium.Cartesian3.subtract(point2, point1, new Cesium.Cartesian3());
    // 计算差向量的长度，即两个点之间的距离
    return Cesium.Cartesian3.magnitude(diff);
}

function analysisAdd() {
    pickPositions = [];
    boardLines = [];
    frustrumLabel = null;
    viewPointFlag = false;
    setBuildFrustrumHandler(true);
}

function analysisRemove() {
    viewer.entities.removeAll();
    viewer.entities.add(tileset);
    pickPositions = [];
    pickPoints = [];
    boardLines = [];
    // 移除事件处理器
    if (handler) {
        handler.destroy();
        handler = null;
    }
    frustrumLabel = null;
    viewPointFlag = false;
}
